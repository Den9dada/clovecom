cmake_minimum_required(VERSION 3.14)
project(clovertls_lua LANGUAGES C)
set(LUA_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../third_party/lua")
file(GLOB LUA_SOURCES CONFIGURE_DEPENDS "${LUA_ROOT}/src/*.c")
list(REMOVE_ITEM LUA_SOURCES "${LUA_ROOT}/src/lua.c" "${LUA_ROOT}/src/luac.c")
add_library(clovertls_lua STATIC ${LUA_SOURCES})
target_include_directories(clovertls_lua PUBLIC "${LUA_ROOT}/src")
if(WIN32)
  target_compile_definitions(clovertls_lua PRIVATE _CRT_SECURE_NO_WARNINGS)
else()
  target_link_libraries(clovertls_lua PUBLIC m ${CMAKE_DL_LIBS})
endif()
add_executable(clovertls_lua_runner runner.c)
target_link_libraries(clovertls_lua_runner PRIVATE clovertls_lua)
if(MSVC)
  target_compile_definitions(clovertls_lua_runner PRIVATE _CRT_SECURE_NO_WARNINGS)
  target_compile_options(clovertls_lua_runner PRIVATE /utf-8)
endif()
